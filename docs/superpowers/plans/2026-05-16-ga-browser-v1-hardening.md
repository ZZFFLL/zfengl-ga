# GA Browser V1 Hardening Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Harden the existing `browser_state` and `browser_action` tools so GA can operate same-origin iframes, dynamic SPA pages, custom controls, overlays, and rich text basics from the user's already-open Chrome session.

**Architecture:** Keep the current v1 tool names and the current TMWebDriver / Chrome extension bridge path. Extend `browser_indexer.py` to return richer metadata and same-origin frame state, then extend `browser_actions.py` to consume that cached metadata for frame-aware actions, verification, and bounded waits without creating a browser-use runtime session.

**Tech Stack:** Python, pytest, existing `TMWebDriver.execute_js(...)` bridge, in-page JavaScript generated by `browser_indexer.py` and `browser_actions.py`, existing GenericAgent tool schema JSON files.

---

## Scope And Non-Negotiables

- Keep `browser_state` and `browser_action`; do not create v2 tool names.
- Do not start Chrome, create a user-data-dir, create a browser-use `BrowserSession`, or migrate login state.
- Do not modify `E:\zfengl-ai-project\browser-use`.
- Same-origin iframe traversal is in scope; cross-origin iframe high-level automation is not.
- `select` remains native `<select>` only; AntD/custom select uses `click -> browser_state -> click option`.
- Table/grid support in this plan is read-only positioning metadata only; cell editing is not part of this plan.
- Screenshot logging, CDP real mouse/key fallback, closed Shadow DOM, recording/replay, and multi-page concurrent browser-use sessions are not part of this plan.

## Success Criteria

- Existing tests for current `browser_state` / `browser_action` behavior keep passing.
- `browser_state` returns old fields unchanged and adds optional metadata fields with stable defaults.
- Same-origin iframe elements appear in the indexed state with `frame_path`, `frame_depth`, `frame_url`, and `frame_title`.
- Indexed `click`, `input`, `select`, `keys`, and `wait_index` can operate cached same-origin iframe elements.
- Mutating actions return `verify_hint`; requested verification fails explicitly with `stage="verify_failed"`.
- New wait actions return success or structured timeout: `wait_dom_stable`, `wait_not_busy`, `wait_enabled`, and `wait_route`.
- Custom controls expose `control_kind` and `action_hints`; attempting native `select` on a custom control fails with `control_unsupported` and actionable hints.
- Contenteditable and same-origin iframe editor bodies can be written through `input` and read back through verification.
- Browser-use SOP teaches the simplified sequencing that avoids the `input -> state -> stale index -> keys` failure loop.

function ownerWindowOf(el) {
  const ownerDocument = el && el.ownerDocument ? el.ownerDocument : document;

function ownerDocumentContains(el) {
  if (!el || !el.ownerDocument) return false;
  return el.ownerDocument.contains(el);
}

function documentForFramePath(framePath) {
  let currentDocument = document;
  for (const step of framePath || []) {
    const frames = Array.from(currentDocument.querySelectorAll("iframe, frame"));
    const frame = frames[Number(step.index)];
    if (!frame) {
      return { error: fail("frame_unavailable", "Frame path is no longer available. Run browser_state again.") };
    }
    try {
      currentDocument = frame.contentDocument;
    } catch (_error) {
      return { error: fail("frame_unavailable", "Frame is not same-origin or no longer readable.") };
    }
    if (!currentDocument) {
      return { error: fail("frame_unavailable", "Frame document is not available.") };
    }
  }
  return { document: currentDocument };
}

```javascript
function readElementValue(el) {
  if (!el) return "";
  const tag = tagOf(el);
  const type = String(el.getAttribute("type") || "").toLowerCase();
  if ("value" in el) {
    return type === "password" && el.value ? "[REDACTED]" : String(el.value || "");
  }
  if (isContentEditableTarget(el)) {
    return String(el.innerText || el.textContent || "");
  }
  return String(el.innerText || el.textContent || "");
}

function verifySuccess(type, observed, expected) {
  return { type, observed, expected, status: "success" };
}

function verifyFailure(type, observed, expected) {
  return {
    status: "failed",
    action: request.action,
    index: request.index,
    stage: "verify_failed",
    error: "Action completed but verification failed.",
    verification: { type, observed, expected, status: "failed" },
    observed,
    expected,
    retryable: true,
  };
}

function verifyAction(el) {
  if (!request.verify) {
    return { verification: null };
  }
  if (request.verify === "field_value") {
    const expected = String(request.verify_value !== null && request.verify_value !== undefined ? request.verify_value : request.text || request.value || "");
    const observed = readElementValue(el);
    if (observed !== expected) return { error: verifyFailure("field_value", observed, expected) };
    return { verification: verifySuccess("field_value", observed, expected) };
  }
  if (request.verify === "element_text") {
    const expected = String(request.verify_text || request.text || "");
    const observed = readElementValue(el);
    if (expected && !observed.includes(expected)) return { error: verifyFailure("element_text", observed, expected) };
    return { verification: verifySuccess("element_text", observed, expected) };
  }
  if (request.verify === "text") {
    const expected = String(request.verify_text || request.text || "");
    const observed = allReadableText();
    if (expected && !observed.includes(expected)) return { error: verifyFailure("text", observed.slice(0, 500), expected) };
    return { verification: verifySuccess("text", expected, expected) };
  }
  if (request.verify === "selector") {
    const expected = String(request.verify_selector || request.selector || "");
    const found = Boolean(expected && queryAllReadableDocuments(expected).length);
    if (!found) return { error: verifyFailure("selector", "missing", expected) };
    return { verification: verifySuccess("selector", "found", expected) };
  }
  return { error: fail("invalid_args", "Unsupported verification type.") };
}
```

function labelsOf(element) {
  const values = [];
  const id = element.getAttribute("id");
  if (id) {
    for (const label of element.ownerDocument.querySelectorAll(`label[for="${cssEscape(id)}"]`)) {
      values.push(textOf(label));
    }
  }
  const ariaLabel = element.getAttribute("aria-label");
  if (ariaLabel) values.push(ariaLabel);
  const labelledBy = element.getAttribute("aria-labelledby");
  if (labelledBy) {
    for (const part of labelledBy.split(/\s+/)) {
      const target = element.ownerDocument.getElementById(part);
      if (target) values.push(textOf(target));
    }
  }
  const parentLabel = element.closest("label");
  if (parentLabel) values.push(textOf(parentLabel));
  const placeholder = element.getAttribute("placeholder");
  if (placeholder) values.push(placeholder);
  return Array.from(new Set(values.filter(Boolean))).map(boundedText);
}

function attributesOf(element) {
  return {
    id: element.getAttribute("id") || "",
    name: element.getAttribute("name") || "",
    class: boundedText(element.getAttribute("class") || ""),
    role: element.getAttribute("role") || "",
    aria_expanded: element.getAttribute("aria-expanded") || "",
    aria_controls: element.getAttribute("aria-controls") || "",
    aria_invalid: element.getAttribute("aria-invalid") || "",
    data_testid: element.getAttribute("data-testid") || element.getAttribute("data-test-id") || "",
    autocomplete: element.getAttribute("autocomplete") || "",
  };
}

function validationOf(element) {
  return {
    required: Boolean(element.required || element.getAttribute("aria-required") === "true"),
    invalid: Boolean(element.matches && element.matches(":invalid")) || element.getAttribute("aria-invalid") === "true",
    read_only: Boolean(element.readOnly || element.getAttribute("aria-readonly") === "true"),
    max_length: element.getAttribute("maxlength") || "",
    pattern: element.getAttribute("pattern") || "",
  };
}

function stableKeyOf(element, tag, role) {
  const attrs = attributesOf(element);
  if (attrs.data_testid) return `${tag}[data-testid="${attrs.data_testid}"]`;
  if (attrs.id) return `${tag}#${attrs.id}`;

  const label = labelsOf(element)[0] || "";

function layerContextOf(element) {
  const root = element.closest("[role='dialog'], [aria-modal='true'], .ant-modal, .ant-drawer, .ant-select-dropdown, .ant-dropdown, .modal, .drawer, .popover, .dropdown");
  if (!root) return { layer: "main", layer_root_hint: "", modal_rank: 0 };
  const text = `${root.className || ""} ${root.getAttribute("role") || ""}`.toLowerCase();
  let layer = "unknown";
  if (text.includes("modal") || root.getAttribute("aria-modal") === "true") layer = "modal";
  else if (text.includes("drawer")) layer = "drawer";
  else if (text.includes("select") || text.includes("dropdown")) layer = "dropdown";
  else if (text.includes("popover")) layer = "popover";
  return { layer, layer_root_hint: selectorHint(root), modal_rank: 1 };
}

function controlKindOf(element, tag, role, type) {
  if (isContentEditableTarget(element)) return "contenteditable";
  if (tag === "select") return "native_select";
  if (tag === "textarea") return "textarea";
  if (tag === "button" || role === "button") return "button";
  if (tag === "a" || role === "link") return "link";
  if (role === "option") return "option";
  if (role === "combobox") return tag === "input" ? "combobox_input" : "custom_select";
  if (tag === "input" && ["date", "datetime-local", "month", "time"].includes(String(type).toLowerCase())) return "date_input";
  if (tag === "input") return "native_input";
  return role || tag;
}

}
